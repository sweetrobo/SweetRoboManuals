QR Code Generation Commands

# Generate all predefined QR codes
npm run qr

# Create a custom QR code
npm run qr:custom -- --url "https://example.com" --id "example" --desc "Example QR"

# Add QR code to database
npm run qr:add -- --url "https://example.com" --id "example" --desc "Example QR" --category "custom"

# Direct command with all options
node scripts/generate-rounded-qr.js --custom --url "URL" --id "ID" --desc "Description" --category "Category" --manuals "manual1,manual2"
